query Posts(
  $schoolId: ID
  $topicId: ID
  $subtopicId: ID
  $keyword: String
  $orderBy: PostOrder
  $after: String
  $excludeTopicId: ID
  $rankFilter: RankFilter
) {
  posts(
    filter: {
      schoolId: $schoolId
      topicId: $topicId
      subtopicId: $subtopicId
      keyword: $keyword
      excludeTopicId: $excludeTopicId
      rankFilter: $rankFilter
    }
    orderBy: $orderBy
    first: 10
    after: $after
  ) {
    edges {
      node {
        ...PostFields
        __typename
      }
      __typename
    }
    totalCount
    pageInfo {
      hasPreviousPage
      hasNextPage
      startCursor
      endCursor
      __typename
    }
    __typename
  }
}
